팀원 시스템이 호출하는 능력치 API

  1. 1LHJ의 아이템 3종이 이 API만 호출
  2. 2속도 버프는 이동 힘의 배율로 반영
  3. 3점수는 스테이지 매니저가 관리하도록 위임
PB.cs
void Update()
{
    if (Input.GetButtonDown("Jump") && !isJump)
    {
        isJump = true;
        rigid.AddForce(Vector3.up * jumpPower, ForceMode.Impulse);
    }
}

void FixedUpdate()
{
    float h = Input.GetAxisRaw("Horizontal");
    float v = Input.GetAxisRaw("Vertical");
    rigid.AddForce(new Vector3(h, 0, v) * playerSpeed, ForceMode.Impulse);
}

void OnCollisionEnter(Collision collision)
{
    if (collision.gameObject.CompareTag("Floor")) isJump = false;
}
